+import { useEffect, useRef } from 'react';
+import ChatMessage from './components/ChatMessage';
+import { useStore } from './store';
+
+function App() {
+  const { messages, addMessages } = useStore();
+  const wsRef = useRef<WebSocket | null>(null);
+  const messagesEndRef = useRef<HTMLDivElement>(null);
+
+  const scrollToBottom = () => {
+    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
+  };
+
+  useEffect(() => {
+    scrollToBottom();
+  }, [messages]);
+
+  useEffect(() => {
+    // Determine the WebSocket URL based on the current origin
+    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
+    const wsUrl = `${protocol}//${window.location.host}/ws`;
+    
+    const ws = new WebSocket(wsUrl);
+    wsRef.current = ws;
+
+    ws.onopen = () => {
+      console.log('Connected to WebSocket');
+    };
+
+    ws.onmessage = (event) => {
+      try {
+        const data = JSON.parse(event.data);
+        if (data.type === 'delete' && data.messageId) {
+          useStore.getState().removeMessage(data.messageId);
+        } else if (Array.isArray(data)) {
+          addMessages(data);
+        } else {
+          addMessages([data]);
+        }
+      } catch (err) {
+        console.error('Failed to parse WebSocket message:', err);
+      }
+    };
+
+    ws.onerror = (err) => {
+      console.error('WebSocket error:', err);
+    };
+
+    ws.onclose = () => {
+      console.log('WebSocket disconnected');
+    };
+
+    return () => {
+      ws.close();
+    };
+  }, [addMessages]);
+
+  return (
+    <div className="h-screen bg-gray-900 text-gray-100 flex flex-col p-4 overflow-hidden">
+      <header className="mb-4 shrink-0">
+        <h1 className="text-xl font-bold text-gray-200">Baba TTS Chat</h1>
+      </header>
+      
+      <main className="flex-1 overflow-y-auto flex flex-col bg-gray-950 rounded border border-gray-800 shadow-inner p-4">
+        {messages.map((msg) => (
+          <ChatMessage key={msg.messageId} message={msg} />
+        ))}
+        <div ref={messagesEndRef} />
+      </main>
+    </div>
+  );
+}
+
+export default App;

+import { create } from 'zustand';
+import type { ChatMessage } from '../../src/shared/types';
+
+interface StoreState {
+  messages: ChatMessage[];
+  addMessages: (newMessages: ChatMessage[]) => void;
+  updateMessageStatus: (messageId: string, status: ChatMessage['status']) => void;
+  removeMessage: (messageId: string) => void;
+}
+
+export const useStore = create<StoreState>((set) => ({
+  messages: [],
+  addMessages: (newMessages) =>
+    set((state) => {
+      // To handle potential duplicates in case we get same message ID again, 
+      // but task just says "addMessages". Let's stick to simple append.
+      const updatedMessages = [...state.messages, ...newMessages];
+      return {
+        messages: updatedMessages.slice(-500),
+      };
+    }),
+  updateMessageStatus: (messageId, status) =>
+    set((state) => ({
+      messages: state.messages.map((msg) =>
+        msg.messageId === messageId ? { ...msg, status } : msg
+      ),
+    })),
+  removeMessage: (messageId) =>
+    set((state) => ({
+      messages: state.messages.filter((msg) => msg.messageId !== messageId),
+    })),
+}));

+import { Hono } from 'hono';
+import { setCookie } from 'hono/cookie';
+import { google } from 'googleapis';
+import { readdirSync, readFileSync } from 'fs';
+import { join } from 'path';

+import { startYouTubePoller } from './youtube.js';
+
+const authRouter = new Hono();
+
+let cachedCreds: { client_id: string; client_secret: string; redirect_uris?: string[] } | null = null;
+
+export function getOAuth2Client() {
+  if (!cachedCreds) {
+    const files = readdirSync(process.cwd()).filter(f => f.startsWith('client_secret_') && f.endsWith('.json'));
+    if (files.length === 0) {
+      throw new Error('No client_secret_*.json found in project root');
+    }
+    
+    const filePath = join(process.cwd(), files[0]);
+    const content = JSON.parse(readFileSync(filePath, 'utf-8'));
+    
+    // Handle both "web" and "installed" types of OAuth credentials
+    cachedCreds = content.web || content.installed;
+    if (!cachedCreds) {
+      throw new Error('Invalid client secret format');
+    }
+  }
+  
+  return new google.auth.OAuth2(
+    cachedCreds.client_id,
+    cachedCreds.client_secret,
+    cachedCreds.redirect_uris?.[0] || 'http://localhost:7060/auth/google/callback'
+  );
+}
+
+authRouter.get('/google', (c) => {
+  const client = getOAuth2Client();
+  const url = client.generateAuthUrl({
+    access_type: 'offline',
+    prompt: 'consent',
+    scope: [
+      'https://www.googleapis.com/auth/youtube.readonly',
+      'https://www.googleapis.com/auth/youtube.force-ssl'
+    ]
+  });
+  return c.redirect(url);
+});
+
+authRouter.get('/google/callback', async (c) => {
+  const code = c.req.query('code');
+  if (!code) {
+    return c.text('No code provided', 400);
+  }
+  
+  const client = getOAuth2Client();
+  const { tokens } = await client.getToken(code);
+  
+  client.setCredentials(tokens);
+  
+  const youtube = google.youtube({
+    version: 'v3',
+    auth: client
+  });
+  
+  let accountId: string = randomUUID();
+  let channelId = null;
+  
+  try {
+    const channelRes = await youtube.channels.list({ part: ['id'], mine: true });
+    if (channelRes.data.items && channelRes.data.items.length > 0) {
+      channelId = channelRes.data.items[0].id;
+      if (channelId) {
+        accountId = channelId; // Use channel ID as the account ID if available
+      }
+    }
+  } catch (err) {
+    console.error('Failed to fetch youtube channel:', err);
+  }
+  
+  if (tokens.refresh_token) {
+    const insertAccount = db.prepare(`
+      INSERT INTO oauth_accounts (id, refresh_token, channel_id) 
+      VALUES (?, ?, ?)
+      ON CONFLICT(id) DO UPDATE SET refresh_token=excluded.refresh_token, channel_id=excluded.channel_id
+    `);
+    insertAccount.run(accountId, tokens.refresh_token, channelId);
+  } else {
+    const exists = db.prepare('SELECT id FROM oauth_accounts WHERE id = ?').get(accountId);
+    if (!exists) {
+      const insertAccount = db.prepare(`
+        INSERT INTO oauth_accounts (id, refresh_token, channel_id)
+        VALUES (?, ?, ?)
+      `);
+      insertAccount.run(accountId, '', channelId);
+    }
+  }
+  
+  const sessionId = randomUUID();
+  db.prepare(`
+    INSERT INTO sessions (id, account_id) VALUES (?, ?)
+  `).run(sessionId, accountId);
+  
+  setCookie(c, 'session', sessionId, {
+    httpOnly: true,
+    secure: true,
+    path: '/',
+    sameSite: 'Lax',
+    maxAge: 60 * 60 * 24 * 7
+  });
+  
+  startYouTubePoller(accountId);
+  
+  return c.text('Authentication successful! You can close this window.');
+});

+import { Hono } from 'hono';
+import { z } from 'zod';
+import { google } from 'googleapis';
+import { getCookie } from 'hono/cookie';
+import { db } from './db.js';
+import { getOAuth2Client } from './auth.js';
+import { broadcastMessageDelete } from './ws.js';
+
+export const moderationRouter = new Hono();
+
+const ModerationBodySchema = z.object({
+  messageId: z.string(),
+  action: z.enum(['delete', 'timeout', 'ban']),
+});
+
+moderationRouter.post('/moderate', async (c) => {
+  const sessionId = getCookie(c, 'session');
+  if (!sessionId) {
+    return c.json({ error: 'Unauthorized' }, 401);
+  }
+
+  const sessionRow = db.prepare('SELECT account_id FROM sessions WHERE id = ?').get(sessionId) as { account_id: string } | undefined;
+  if (!sessionRow) {
+    return c.json({ error: 'Unauthorized' }, 401);
+  }
+
+  const accountId = sessionRow.account_id;
+  const accountRow = db.prepare('SELECT refresh_token FROM oauth_accounts WHERE id = ?').get(accountId) as { refresh_token: string } | undefined;
+  if (!accountRow || !accountRow.refresh_token) {
+    return c.json({ error: 'OAuth account not found or not connected' }, 401);
+  }
+
+  let body;
+  try {
+    const rawBody = await c.req.json();
+    body = ModerationBodySchema.parse(rawBody);
+  } catch (error) {
+    return c.json({ error: 'Invalid request body' }, 400);
+  }
+
+  const { messageId, action } = body;
+  
+  const cached = db.prepare('SELECT live_chat_id, author_channel_id FROM messages WHERE id = ?').get(messageId) as { live_chat_id: string; author_channel_id: string } | undefined;
+
+  if (action === 'timeout' || action === 'ban') {
+    if (!cached) {
+      return c.json({ error: 'Message details not found in cache to perform ban' }, 404);
+    }
+  }
+
+  const client = getOAuth2Client();
+  client.setCredentials({ refresh_token: accountRow.refresh_token });
+  const youtube = google.youtube({ version: 'v3', auth: client });
+
+  try {
+    if (action === 'delete') {
+      await youtube.liveChatMessages.delete({ id: messageId });
+    } else if (action === 'timeout' || action === 'ban') {
+      const type = action === 'timeout' ? 'temporary' : 'permanent';
+      const banDurationSeconds = action === 'timeout' ? 300 : undefined; // 5 minutes
+
+      await youtube.liveChatBans.insert({
+        part: ['snippet'],
+        requestBody: {
+          snippet: {
+            liveChatId: cached!.live_chat_id,
+            type,
+            banDurationSeconds: banDurationSeconds ? String(banDurationSeconds) : undefined,
+            bannedUserDetails: {
+              channelId: cached!.author_channel_id
+            }
+          }
+        }
+      });
+    }
+
+    broadcastMessageDelete(messageId);
+
+    return c.json({ success: true });
+  } catch (error: any) {
+    console.error('Moderation action failed:', error);
+    return c.json({ error: 'Moderation action failed', details: error.message }, 500);
+  }
+});

+import { WebSocketServer, WebSocket } from 'ws';
+import type { Server } from 'node:http';
+import type { ChatMessage } from '../shared/types.js';
+
+let wss: WebSocketServer | null = null;
+
+export function initWebSocketServer(server: Server) {
+  wss = new WebSocketServer({ server });
+
+  wss.on('connection', (ws) => {
+    console.log('Client connected to WebSocket');
+    ws.on('error', console.error);
+    ws.on('close', () => {
+      console.log('Client disconnected');
+    });
+  });
+}
+
+export function broadcastMessages(msgs: ChatMessage[]) {
+  if (!wss) {
+    console.warn('WebSocket server not initialized');
+    return;
+  }
+  
+  if (msgs.length === 0) return;
+  
+  const data = JSON.stringify(msgs);
+  wss.clients.forEach((client) => {
+    if (client.readyState === WebSocket.OPEN) {
+      client.send(data);
+    }
+  });
+}
+
+export function broadcastMessageDelete(messageId: string) {
+  if (!wss) {
+    console.warn('WebSocket server not initialized');
+    return;
+  }
+  
+  const data = JSON.stringify({ type: 'delete', messageId });
+  wss.clients.forEach((client) => {
+    if (client.readyState === WebSocket.OPEN) {
+      client.send(data);
+    }
+  });
+}

+import { google } from 'googleapis';
+import { db } from './db.js';
+import { getOAuth2Client } from './auth.js';
+import { broadcastMessages } from './ws.js';
+import { enqueueMessageForTTS } from './tts.js';
+import type { ChatMessage } from '../shared/types.js';

+const activePollers = new Set<string>();
+const pollerTimeouts = new Map<string, NodeJS.Timeout>();
+
+function escapeHtml(unsafe: string) {
+  return unsafe
+    .replace(/&/g, "&amp;")
+    .replace(/</g, "&lt;")
+    .replace(/>/g, "&gt;")
+    .replace(/"/g, "&quot;")
+    .replace(/'/g, "&#039;");
+}

+export function _testSetActivePoller(accountId: string) {
+  activePollers.add(accountId);
+}

+export async function startYouTubePoller(accountId: string) {
+  if (activePollers.has(accountId)) return;
+  
+  const row = db.prepare('SELECT refresh_token FROM oauth_accounts WHERE id = ?').get(accountId) as { refresh_token: string } | undefined;
+  if (!row || !row.refresh_token) {
+    console.error('No refresh token found for account:', accountId);
+    return;
+  }
+  
+  const client = getOAuth2Client();
+  client.setCredentials({ refresh_token: row.refresh_token });
+  
+  const youtube = google.youtube({ version: 'v3', auth: client });
+  
+  try {
+    const broadcastRes = await youtube.liveBroadcasts.list({
+      part: ['snippet'],
+      broadcastStatus: 'active',
+      broadcastType: 'all'
+    });
+    
+    const items = broadcastRes.data.items;
+    if (!items || items.length === 0) {
+      console.log('No active broadcasts found for account:', accountId);
+      return;
+    }
+    
+    const liveChatId = items[0].snippet?.liveChatId;
+    if (!liveChatId) {
+      console.log('No live chat ID found for the broadcast.');
+      return;
+    }
+    
+    activePollers.add(accountId);
+    pollLiveChat(accountId, youtube, liveChatId, '');
+    
+  } catch (error) {
+    console.error(`Error starting YouTube poller for account ${accountId}:`, error);
+  }
+}
+
+
+export async function pollLiveChat(accountId: string, youtube: any, liveChatId: string, pageToken: string) {
+  if (!activePollers.has(accountId)) return;
+  
+  try {
+    const chatRes = await youtube.liveChatMessages.list({
+      liveChatId,
+      part: ['snippet', 'authorDetails'],
+      pageToken: pageToken || undefined,
+    });
+    
+    const nextToken = chatRes.data.nextPageToken || '';
+    const pollingInterval = chatRes.data.pollingIntervalMillis || 5000;
+    
+    const items = chatRes.data.items || [];
+    const messages: ChatMessage[] = items.map((item: any) => {
+      const msgId = item.id || '';
+      if (msgId) {
+        try {
+          db.prepare(`
+            INSERT OR IGNORE INTO messages (id, live_chat_id, author_channel_id)
+            VALUES (?, ?, ?)
+          `).run(msgId, liveChatId, item.snippet?.authorChannelId || '');
+        } catch (err) {
+          console.error('Failed to store message metadata:', err);
+        }
+      }
+      return {
+        messageId: msgId,
+        authorName: escapeHtml(item.authorDetails?.displayName || 'Unknown'),
+        text: escapeHtml(item.snippet?.displayMessage || ''),
+        status: 'default'
+      };
+    });
+    
+    if (messages.length > 0) {
+      broadcastMessages(messages);
+      messages.forEach(msg => enqueueMessageForTTS(msg));
+    }
+    
+    const timeout = setTimeout(() => pollLiveChat(accountId, youtube, liveChatId, nextToken), pollingInterval);
+    pollerTimeouts.set(accountId, timeout);
+    
+  } catch (error: any) {
+    console.error(`Error polling live chat for account ${accountId}:`, error.message || error);
+    
+    const statusCode = error.code || error.status;
+    
+    // Non-retriable errors
+    if (statusCode === 401 || statusCode === 403 || statusCode === 404) {
+      console.error(`Fatal error ${statusCode}. Stopping poller for account ${accountId}.`);
+      activePollers.delete(accountId);
+      pollerTimeouts.delete(accountId);
+      return;
+    }
+    
+    let retryWait = 5000;
+    if (statusCode === 429 || statusCode === 500) {
+      console.log('Rate limit or server error. Waiting 10 seconds before retrying...');
+      retryWait = 10000;
+    }
+    
+    // Retry with the same pageToken
+    const timeout = setTimeout(() => pollLiveChat(accountId, youtube, liveChatId, pageToken), retryWait);
+    pollerTimeouts.set(accountId, timeout);
+  }
+}
+
+export function stopYouTubePoller(accountId?: string) {
+  if (accountId) {
+    activePollers.delete(accountId);
+    const timeout = pollerTimeouts.get(accountId);
+    if (timeout) {
+      clearTimeout(timeout);
+      pollerTimeouts.delete(accountId);
+    }
+  } else {
+    activePollers.clear();
+    for (const timeout of pollerTimeouts.values()) {
+      clearTimeout(timeout);
+    }
+    pollerTimeouts.clear();
+  }
